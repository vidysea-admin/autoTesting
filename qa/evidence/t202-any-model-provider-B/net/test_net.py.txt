import socket, asyncio, time, errno
import pytest

def classify(label, fn):
    t=time.time()
    try:
        r = fn()
        out = f"RESULT-NO-EXC {r!r}"
        verdict = "REACHED-THE-WIRE?" 
    except AssertionError as e:
        out, verdict = f"AssertionError: {str(e)[:70]}", "REFUSED"
    except BaseException as e:
        out = f"{type(e).__name__}: {str(e)[:90]}"
        verdict = "REACHED-THE-WIRE" if isinstance(e,(TimeoutError,socket.timeout,ConnectionError,OSError,BlockingIOError)) and "tried a real" not in str(e) else "OTHER"
    print(f"RES|{label}|{verdict}|{time.time()-t:.2f}s|{out}")

def test_all(no_network):
    T=("192.0.2.1",9)
    def raw_connect(fam=socket.AF_INET, addr=T, typ=socket.SOCK_STREAM):
        s=socket.socket(fam,typ); s.settimeout(0.5)
        try: s.connect(addr)
        finally: s.close()
    def ex():
        s=socket.socket(); s.settimeout(0.5)

        finally: s.close()
    def udp_connect():
        s=socket.socket(socket.AF_INET,socket.SOCK_DGRAM)

        finally: s.close()
