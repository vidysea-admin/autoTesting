def _show(tag: str, ctx: Any, spy: Spies, name: str, args: Any, rawsecret: str = RAW) -> None:
    try:
        out = tools.dispatch(ctx, name, args)
    except BaseException as exc:  # noqa
        txt = f"{type(exc).__name__}: {exc}"
        print(f"PB {tag:38} RAISED {txt[:110]!r} secret_in_exc={rawsecret in txt} sessions={spy.sessions} calls={sum(len(v) for v in spy.calls.values())}")
        return
    j = out.model_dump_json()
    print(f"PB {tag:38} refusal={out.refusal} ok={out.ok} raw_in_out={rawsecret in j} summary={out.summary[:70]!r} sessions={spy.sessions} calls={sum(len(v) for v in spy.calls.values())}")


def test_encodings(tmp_path: Path, spies: Spies) -> None:
    ctx = _ctx(tmp_path, spies)
    _grant(ctx.store, ApprovalKind.LIVE_CASE)
    b64 = base64.b64encode(RAW.encode()).decode()
    b64u = base64.urlsafe_b64encode(RAW.encode()).decode().rstrip("=")
    hx = binascii.hexlify(RAW.encode()).decode()
    pct = urllib.parse.quote(RAW, safe="")
    pct_all = "".join(f"%{b:02x}" for b in RAW.encode())
    rev = RAW[::-1]
    upper = RAW.upper()
    spaced = " ".join(RAW)
    dotted = ".".join(RAW)
    half1, half2 = RAW[: len(RAW) // 2], RAW[len(RAW) // 2:]
    forms = {"raw": RAW, "b64": b64, "b64url": b64u, "hex": hx, "pct": pct, "pct_all": pct_all,
             "rev": rev, "upper": upper, "spaced": spaced, "dotted": dotted,
             "rot13": __import__("codecs").encode(RAW, "rot13"), "b32": base64.b32encode(RAW.encode()).decode()}
    for tag, v in forms.items():
        _show(f"case_id={tag}", ctx, spies, "run_case", {"case_id": v})
    _show("split across two args (extra)", ctx, spies, "run_case", {"case_id": half1, "x": half2})
    _show("split case_id|case_id concatenated name", ctx, spies, half1 + "run_case", {"case_id": half2})
    _show("nested dict arg", ctx, spies, "run_case", {"case_id": {"a": [RAW]}})
    _show("dict KEY raw", ctx, spies, "run_case", {RAW: 1})
    _show("dict KEY b64", ctx, spies, "run_case", {b64: 1})
    _show("dict KEY rev", ctx, spies, "run_case", {"case_id": "c", rev: 1})
    _show("tool name raw", ctx, spies, RAW, {})
    _show("tool name b64", ctx, spies, b64, {})
    _show("crawl bool/num fields", ctx, spies, "crawl", {"max_actions": RAW})
    _show("crawl nested list", ctx, spies, "crawl", {"max_actions": [RAW]})

def test_exception_escape(tmp_path: Path, spies: Spies, monkeypatch: pytest.MonkeyPatch) -> None:
    ctx = _ctx(tmp_path, spies)
    _grant(ctx.store, ApprovalKind.LIVE_CASE)
    _grant(ctx.store, ApprovalKind.CRAWL)
    case = ctx.store.add_case(_case([_nav()]))
    def boom(c: Any, s: Any) -> Any:
        raise RuntimeError(f"driver died typing {RAW}")
    monkeypatch.setattr(tools.execute, "run_case", boom)
    _show("stage raises w/ secret", ctx, spies, "run_case", {"case_id": case.id})
    def boomc(*a: Any, **k: Any) -> Any:
        raise RuntimeError(f"crawl died {RAW}")
    monkeypatch.setattr(tools.explore, "run_crawl", boomc)
    _show("crawl stage raises w/ secret", ctx, spies, "crawl", {})
    def bad_open(obs: Any) -> Any:
        raise OSError(f"cannot launch with {RAW}")
    ctx.open_session = bad_open
    _show("open_session raises", ctx, spies, "run_case", {"case_id": case.id})
