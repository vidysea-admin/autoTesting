            rows.append((pol.value, label, "REFUSED" if r else "ALLOWED", r))

        for label, *_ in [(l,) for l in cases]:

def test_al3c_in_dispatch_allowed_enter(tmp_path: Path, spies: Spies) -> None:
    ctx = _ctx(tmp_path, spies, WritePolicy.READ_ONLY)
    _grant(ctx.store, ApprovalKind.LIVE_CASE)
    for lab, st in (("enter", Step(order=2, action=Action.PRESS_KEY, target="Comment", value="Enter")),
                    ("nav-delete", Step(order=2, action=Action.NAVIGATE, target="/account/delete?id=1")),
                    ("click-save", Step(order=2, action=Action.CLICK, target="Save"))):
        c = ctx.store.add_case(_case([_nav(), st]))
        out = _go(ctx, "run_case", {"case_id": c.id})
        print("DISPATCH", lab, out.ok, out.refusal, "run_case calls:", spies.count("run_case"))
